﻿using ASKART.API.Data;
using ASKART.API.DTOs;
using ASKART.API.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace ASKART.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class ChatMembersController : ControllerBase
{
    private readonly AskartContext _context;

    public ChatMembersController(AskartContext context)
    {
        _context = context;
    }

    [HttpPost]
    public async Task<IActionResult> AddMember(AddChatMemberDto dto)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == dto.ChatId);

        if (!chatExists)
        {
            return NotFound(new
            {
                message = "Чат не найден."
            });
        }

        var userExists = await _context.Users
            .AnyAsync(u => u.IdUser == dto.UserId);

        if (!userExists)
        {
            return NotFound(new
            {
                message = "Пользователь не найден."
            });
        }

        var memberExists = await _context.Chatmembers
            .AnyAsync(cm =>
                cm.ChatId == dto.ChatId &&
                cm.UserId == dto.UserId);

        if (memberExists)
        {
            return BadRequest(new
            {
                message = "Пользователь уже состоит в этом чате."
            });
        }

        var chatMember = new Chatmember
        {
            ChatId = dto.ChatId,
            UserId = dto.UserId,
            Role = dto.Role
        };

        _context.Chatmembers.Add(chatMember);
        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Пользователь добавлен в чат.",
            chatId = chatMember.ChatId,
            userId = chatMember.UserId,
            role = chatMember.Role
        });
    }


    [HttpGet("{chatId}")]
    public async Task<IActionResult> GetMembers(int chatId)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == chatId);

        if (!chatExists)
        {
            return NotFound(new
            {
                message = "Чат не найден."
            });
        }

        var members = await _context.Chatmembers
            .Where(cm => cm.ChatId == chatId)
            .Include(cm => cm.User)
            .Select(cm => new
            {
                userId = cm.UserId,
                surname = cm.User.Surname,
                name = cm.User.Name,
                lastname = cm.User.Lastname,
                login = cm.User.Login,
                role = cm.Role
            })
            .ToListAsync();

        return Ok(members);
    }


    [HttpDelete("{chatId}/{userId}")]
    public async Task<IActionResult> RemoveMember(int chatId, int userId)
    {
        var member = await _context.Chatmembers
            .FirstOrDefaultAsync(cm =>
                cm.ChatId == chatId &&
                cm.UserId == userId);

        if (member == null)
        {
            return NotFound(new
            {
                message = "Пользователь не состоит в этом чате."
            });
        }

        _context.Chatmembers.Remove(member);
        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Пользователь удалён из чата."
        });
    }
}

﻿using ASKART.API.Data;
using ASKART.API.DTOs;
using ASKART.API.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace ASKART.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class ChatsController : ControllerBase
{
    private readonly AskartContext _context;

    public ChatsController(AskartContext context)
    {
        _context = context;
    }

    [HttpPost("private")]
    public async Task<IActionResult> CreatePrivateChat(CreatePrivateChatDto dto)
    {
        var user = await _context.Users
            .FirstOrDefaultAsync(u => u.IdUser == dto.UserId);

        if (user == null)
        {
            return NotFound(new
            {
                message = "Пользователь не найден."
            });
        }

        var otherUser = await _context.Users
            .FirstOrDefaultAsync(u => u.IdUser == dto.OtherUserId);

        if (otherUser == null)
        {
            return NotFound(new
            {
                message = "Второй пользователь не найден."
            });
        }

        if (dto.UserId == dto.OtherUserId)
        {
            return BadRequest(new
            {
                message = "Нельзя создать личный чат с самим собой."
            });
        }

        var existingChatId = await _context.Chats
    .Where(c => c.TypeChat == "Личный")
    .Where(c => _context.Chatmembers.Any(cm =>
        cm.ChatId == c.IdChat &&
        cm.UserId == dto.UserId))
    .Where(c => _context.Chatmembers.Any(cm =>
        cm.ChatId == c.IdChat &&
        cm.UserId == dto.OtherUserId))
    .Select(c => (int?)c.IdChat)
    .FirstOrDefaultAsync();

        if (existingChatId != null)
        {
            return Ok(new
            {
                message = "Личный чат уже существует.",
                chatId = existingChatId
            });
        }

        var lastId = await _context.Chats
            .Select(c => (int?)c.IdChat)
            .MaxAsync() ?? 0;

        var chat = new Chat
        {
            IdChat = lastId + 1,
            Name = $"{otherUser.Surname} {otherUser.Name}",
            TypeChat = "Личный",
            TypeGroup = null,
            UserId = dto.UserId
        };

        _context.Chats.Add(chat);

        var firstMember = new Chatmember
        {
            ChatId = chat.IdChat,
            UserId = dto.UserId,
            Role = "Участник"
        };

        var secondMember = new Chatmember
        {
            ChatId = chat.IdChat,
            UserId = dto.OtherUserId,
            Role = "Участник"
        };

        _context.Chatmembers.Add(firstMember);
        _context.Chatmembers.Add(secondMember);

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Личный чат создан.",
            chatId = chat.IdChat,
            name = chat.Name,
            typeChat = chat.TypeChat,
            members = new[]
            {
                new
                {
                    userId = firstMember.UserId,
                    role = firstMember.Role
                },
                new
                {
                    userId = secondMember.UserId,
                    role = secondMember.Role
                }
            }
        });
    }


    [HttpPost("group")]
    public async Task<IActionResult> CreateGroup(CreateGroupChatDto dto)

        var creatorExists = await _context.Users

            IdChat = lastId + 1,
            Name = dto.Name,
            TypeChat = "Группа",
            TypeGroup = dto.TypeGroup,
            UserId = dto.CreatorId

        _context.Chats.Add(chat);

        var creatorMember = new Chatmember
        {
            ChatId = chat.IdChat,
            UserId = dto.CreatorId,

            chatId = chat.IdChat,

                userId = creatorMember.UserId,
                role = creatorMember.Role
            }

    [HttpGet("user/{userId}")]
    public async Task<IActionResult> GetUserChats(int userId)
    {
        var userExists = await _context.Users
            .AnyAsync(u => u.IdUser == userId);

        if (!userExists)
        {
            return NotFound(new
            {
                message = "Пользователь не найден."
            });
        }

        var memberships = await _context.Chatmembers
            .Where(cm => cm.UserId == userId)
            .Include(cm => cm.Chat)
            .ToListAsync();

        if (memberships.Count == 0)
        {
            return Ok(new List<object>());
        }

        var chatIds = memberships
            .Select(cm => cm.ChatId)
            .ToList();

        var allMembers = await _context.Chatmembers
            .Where(cm => chatIds.Contains(cm.ChatId))
            .Include(cm => cm.User)
            .ToListAsync();

        var chats = memberships.Select(cm =>
        {
            var chat = cm.Chat;

            if (chat.TypeChat == "Личный")
            {
                var otherMember = allMembers
                    .FirstOrDefault(member =>
                        member.ChatId == chat.IdChat &&
                        member.UserId != userId);

                return new
                {
                    chatId = chat.IdChat,
                    name = otherMember != null
                        ? $"{otherMember.User.Surname} {otherMember.User.Name}"
                        : "Неизвестный пользователь",
                    typeChat = chat.TypeChat,
                    typeGroup = chat.TypeGroup
                };
            }

            return new
            {
                chatId = chat.IdChat,
                name = chat.Name,
                typeChat = chat.TypeChat,
                typeGroup = chat.TypeGroup
            };
        }).ToList();

        return Ok(chats);
    }

}

using ASKART.API.Data;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace ASKART.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class KanbanController : ControllerBase
{
    private readonly AskartContext _context;

    public KanbanController(AskartContext context)
    {
        _context = context;
    }

    [HttpGet("chat/{chatId}")]
    public async Task<IActionResult> GetColumns(int chatId)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == chatId);

        if (!chatExists)
        {
            return NotFound(new { message = "Чат не найден." });
        }

        var columns = await _context.KanbanColumns
            .Where(c => c.ChatId == chatId)
            .OrderBy(c => c.Position)
            .Select(c => new
            {
                columnId = c.IdColumn,
                chatId = c.ChatId,
                name = c.Name,
                color = c.Color,
                position = c.Position
            })
            .ToListAsync();

        return Ok(columns);
    }

    [HttpPost("chat/{chatId}/initialize")]
    public async Task<IActionResult> InitializeBoard(int chatId)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == chatId);

        if (!chatExists)
        {
            return NotFound(new { message = "Чат не найден." });
        }

        var defaults = new[]
        {
            new { Name = "Новая", Color = "#9E9E9E", Position = 1 },
            new { Name = "В работе", Color = "#5DD4FF", Position = 2 },
            new { Name = "Выполнена", Color = "#4CAF50", Position = 3 }
        };

        var existingColumns = await _context.KanbanColumns
            .Where(c => c.ChatId == chatId)
            .ToListAsync();

        var added = new List<object>();

        foreach (var item in defaults)
        {
            if (existingColumns.Any(c => c.Name == item.Name))
            {
                continue;
            }

            var lastId = existingColumns
                .Select(c => c.IdColumn)
                .DefaultIfEmpty(0)
                .Max();

            var column = new ASKART.API.Models.KanbanColumn
            {
                IdColumn = lastId + 1,
                ChatId = chatId,
                Name = item.Name,
                Color = item.Color,
                Position = item.Position
            };

            _context.KanbanColumns.Add(column);
            existingColumns.Add(column);

            added.Add(new
            {
                columnId = column.IdColumn,
                name = column.Name,
                color = column.Color,
                position = column.Position
            });
        }

        if (added.Count == 0)
        {
            return Ok(new
            {
                message = "Стандартные колонки уже существуют.",
                chatId
            });
        }

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Колонки Kanban созданы.",
            chatId,
            columns = added
        });
    }

    [HttpPut("{chatId}/{columnId}")]
    public async Task<IActionResult> UpdateColumn(
        int chatId,
        int columnId,
        UpdateKanbanColumnDto dto)
    {
        var column = await _context.KanbanColumns
            .FirstOrDefaultAsync(c =>
                c.ChatId == chatId &&
                c.IdColumn == columnId);

        if (column == null)
        {
            return NotFound(new
            {
                message = "Колонка не найдена."
            });
        }

        if (string.IsNullOrWhiteSpace(dto.Color) ||
            dto.Color.Length > 20)
        {
            return BadRequest(new
            {
                message = "Укажите корректный цвет длиной до 20 символов."
            });
        }

        if (dto.Position < 1)
        {
            return BadRequest(new
            {
                message = "Позиция колонки должна быть не меньше 1."
            });
        }

        column.Color = dto.Color;
        column.Position = dto.Position;

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Колонка обновлена.",
            columnId = column.IdColumn,
            chatId = column.ChatId,
            name = column.Name,
            color = column.Color,
            position = column.Position
        });
    }
}

public class UpdateKanbanColumnDto
{
    public string Color { get; set; } = null!;

    public int Position { get; set; }
}

using ASKART.API.Data;
using ASKART.API.DTOs;
using ASKART.API.Models;
using ASKART.API.Hubs;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.SignalR;
using Microsoft.EntityFrameworkCore;

namespace ASKART.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class MessagesController : ControllerBase
{
    private readonly AskartContext _context;
    private readonly IHubContext<ChatHub> _hubContext;

    public MessagesController(
        AskartContext context,
        IHubContext<ChatHub> hubContext)
    {
        _context = context;
        _hubContext = hubContext;
    }

    [HttpPost]
    public async Task<IActionResult> SendMessage(SendMessageDto dto)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == dto.ChatId);

        if (!chatExists)
        {
            return NotFound(new
            {
                message = "Чат не найден."
            });
        }

        var userExists = await _context.Users
            .AnyAsync(u => u.IdUser == dto.SenderId);

        if (!userExists)
        {
            return NotFound(new
            {
                message = "Пользователь не найден."
            });
        }

        var isMember = await _context.Chatmembers
            .AnyAsync(cm =>
                cm.ChatId == dto.ChatId &&
                cm.UserId == dto.SenderId);

        if (!isMember)
        {
            return BadRequest(new
            {
                message = "Пользователь не состоит в этом чате."
            });
        }

        if (dto.ReplyId != null)
        {
            var replyExists = await _context.Messages
                .AnyAsync(m =>
                    m.IdMessage == dto.ReplyId &&
                    m.ChatId == dto.ChatId);

            if (!replyExists)
            {
                return BadRequest(new
                {
                    message = "Сообщение, на которое выполняется ответ, не найдено."
                });
            }
        }

        var lastId = await _context.Messages
            .Select(m => (int?)m.IdMessage)
            .MaxAsync() ?? 0;

        var message = new Message
        {
            IdMessage = lastId + 1,
            Text = dto.Text,
            ChatId = dto.ChatId,
            SenderId = dto.SenderId,
            ReplyId = dto.ReplyId,
            SendTime = DateTime.Now
        };

        _context.Messages.Add(message);

        await _context.SaveChangesAsync();

        var sender = await _context.Users
            .FirstAsync(u => u.IdUser == dto.SenderId);

        var response = new
        {
            messageId = message.IdMessage,
            text = message.Text,
            chatId = message.ChatId,
            senderId = message.SenderId,
            senderName = $"{sender.Surname} {sender.Name}",
            replyId = message.ReplyId,
            sendTime = message.SendTime
        };

        await _hubContext.Clients
            .Group($"chat_{message.ChatId}")
            .SendAsync("ReceiveMessage", response);

        return Ok(response);
    }

    [HttpGet("chat/{chatId}")]
    public async Task<IActionResult> GetChatMessages(int chatId)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == chatId);

        if (!chatExists)
        {
            return NotFound(new
            {
                message = "Чат не найден."
            });
        }

        var messages = await _context.Messages
            .Where(m => m.ChatId == chatId)
            .Include(m => m.Sender)
            .OrderBy(m => m.SendTime)
            .Select(m => new
            {
                messageId = m.IdMessage,
                text = m.Text,
                chatId = m.ChatId,
                senderId = m.SenderId,
                senderName = $"{m.Sender.Surname} {m.Sender.Name}",
                replyId = m.ReplyId,
                sendTime = m.SendTime,
                editTime = m.EditTime
            })
            .ToListAsync();

        return Ok(messages);
    }

    [HttpPut("{messageId}")]
    public async Task<IActionResult> EditMessage(
        int messageId,
        [FromBody] string text)
    {
        var message = await _context.Messages
            .FirstOrDefaultAsync(m => m.IdMessage == messageId);

        if (message == null)
        {
            return NotFound(new
            {
                message = "Сообщение не найдено."
            });
        }

        if (string.IsNullOrWhiteSpace(text))
        {
            return BadRequest(new
            {
                message = "Текст сообщения не может быть пустым."
            });
        }

        if (text.Length > 200)
        {
            return BadRequest(new
            {
                message = "Сообщение не должно превышать 200 символов."
            });
        }

        message.Text = text;
        message.EditTime = DateTime.Now;

        await _context.SaveChangesAsync();

        return Ok(new
        {
            messageId = message.IdMessage,
            text = message.Text,
            editTime = message.EditTime
        });
    }
}

using ASKART.API.Data;
using ASKART.API.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace ASKART.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class TaskAssigneesController : ControllerBase
{
    private readonly AskartContext _context;

    public TaskAssigneesController(AskartContext context)
    {
        _context = context;
    }

    [HttpPost]
    public async Task<IActionResult> AddAssignee(
        [FromBody] TaskAssigneeDto dto)
    {
        var task = await _context.Tasks
            .Include(t => t.Users)
            .FirstOrDefaultAsync(t => t.IdTask == dto.TaskId);

        if (task == null)
        {
            return NotFound(new { message = "Задача не найдена." });
        }

        var user = await _context.Users
            .FirstOrDefaultAsync(u => u.IdUser == dto.UserId);

        if (user == null)
        {
            return NotFound(new { message = "Пользователь не найден." });
        }

        var isMember = await _context.Chatmembers
            .AnyAsync(cm =>
                cm.ChatId == task.ChatId &&
                cm.UserId == dto.UserId);

        if (!isMember)
        {
            return BadRequest(new
            {
                message = "Исполнитель не состоит в чате задачи."
            });
        }

        if (task.Users.Any(u => u.IdUser == dto.UserId))
        {
            return BadRequest(new
            {
                message = "Пользователь уже назначен на эту задачу."
            });
        }

        task.Users.Add(user);

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Исполнитель назначен.",
            taskId = task.IdTask,
            userId = user.IdUser,
            surname = user.Surname,
            name = user.Name
        });
    }

    [HttpGet("task/{taskId}")]
    public async Task<IActionResult> GetAssignees(int taskId)
    {
        var task = await _context.Tasks
            .Include(t => t.Users)
            .FirstOrDefaultAsync(t => t.IdTask == taskId);

        if (task == null)
        {
            return NotFound(new { message = "Задача не найдена." });
        }

        var assignees = task.Users
            .Select(u => new
            {
                userId = u.IdUser,
                surname = u.Surname,
                name = u.Name,
                lastname = u.Lastname,
                login = u.Login
            })
            .ToList();

        return Ok(assignees);
    }

    [HttpDelete("{taskId}/{userId}")]
    public async Task<IActionResult> RemoveAssignee(
        int taskId,
        int userId)
    {
        var task = await _context.Tasks
            .Include(t => t.Users)
            .FirstOrDefaultAsync(t => t.IdTask == taskId);

        if (task == null)
        {
            return NotFound(new { message = "Задача не найдена." });
        }

        var user = task.Users
            .FirstOrDefault(u => u.IdUser == userId);

        if (user == null)
        {
            return NotFound(new
            {
                message = "Пользователь не назначен на эту задачу."
            });
        }

        task.Users.Remove(user);

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Исполнитель снят с задачи.",
            taskId,
            userId
        });
    }
}

using ASKART.API.Data;
using ASKART.API.DTOs;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using TaskModel = ASKART.API.Models.Task;

namespace ASKART.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class TasksController : ControllerBase
{
    private readonly AskartContext _context;

    public TasksController(AskartContext context)
    {
        _context = context;
    }

    [HttpPost]
    public async Task<IActionResult> CreateTask(CreateTaskDto dto)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == dto.ChatId);

        if (!chatExists)
        {
            return NotFound(new { message = "Чат не найден." });
        }

        var creatorExists = await _context.Users
            .AnyAsync(u => u.IdUser == dto.CreatorId);

        if (!creatorExists)
        {
            return NotFound(new { message = "Создатель задачи не найден." });
        }

        if (dto.AssigneUserId.HasValue)
        {
            var assigneeExists = await _context.Users
                .AnyAsync(u => u.IdUser == dto.AssigneUserId.Value);

            if (!assigneeExists)
            {
                return NotFound(new { message = "Исполнитель не найден." });
            }
        }

        if (dto.Priority != "Низкий" &&
            dto.Priority != "Средний" &&
            dto.Priority != "Высокий")
        {
            return BadRequest(new
            {
                message = "Приоритет должен быть: Низкий, Средний или Высокий."
            });
        }

        var creatorIsMember = await _context.Chatmembers
            .AnyAsync(cm =>
                cm.ChatId == dto.ChatId &&
                cm.UserId == dto.CreatorId);

        if (!creatorIsMember)
        {
            return BadRequest(new
            {
                message = "Создатель не состоит в этом чате."
            });
        }

        if (dto.AssigneUserId.HasValue)
        {
            var assigneeIsMember = await _context.Chatmembers
                .AnyAsync(cm =>
                    cm.ChatId == dto.ChatId &&
                    cm.UserId == dto.AssigneUserId.Value);

            if (!assigneeIsMember)
            {
                return BadRequest(new
                {
                    message = "Исполнитель не состоит в этом чате."
                });
            }
        }

        var lastId = await _context.Tasks
            .Select(t => (int?)t.IdTask)
            .MaxAsync() ?? 0;

        var task = new TaskModel
        {
            IdTask = lastId + 1,
            ChatId = dto.ChatId,
            Name = dto.Name,
            Describe = dto.Describe,
            Status = "Новая",
            Priority = dto.Priority,
            Deadline = dto.Deadline ?? DateTime.MinValue,
            CreatorId = dto.CreatorId,
            AssigneUserId = dto.AssigneUserId ?? 0
        };

        _context.Tasks.Add(task);

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Задача создана.",
            taskId = task.IdTask,
            chatId = task.ChatId,
            name = task.Name,
            describe = task.Describe,
            status = task.Status,
            priority = task.Priority,
            deadline = task.Deadline,
            creatorId = task.CreatorId,
            assigneUserId = task.AssigneUserId
        });
    }

    [HttpGet("chat/{chatId}")]
    public async Task<IActionResult> GetChatTasks(int chatId)
    {
        var chatExists = await _context.Chats
            .AnyAsync(c => c.IdChat == chatId);

        if (!chatExists)
        {
            return NotFound(new { message = "Чат не найден." });
        }

        var tasks = await _context.Tasks
            .Where(t => t.ChatId == chatId)
            .OrderBy(t => t.IdTask)
            .Select(t => new
            {
                taskId = t.IdTask,
                chatId = t.ChatId,
                name = t.Name,
                describe = t.Describe,
                status = t.Status,
                priority = t.Priority,
                deadline = t.Deadline,
                creatorId = t.CreatorId,
                assigneUserId = t.AssigneUserId
            })
            .ToListAsync();

        return Ok(tasks);
    }

    [HttpGet("{taskId}")]
    public async Task<IActionResult> GetTask(int taskId)
    {
        var task = await _context.Tasks
            .Where(t => t.IdTask == taskId)
            .Select(t => new
            {
                taskId = t.IdTask,
                chatId = t.ChatId,
                name = t.Name,
                describe = t.Describe,
                status = t.Status,
                priority = t.Priority,
                deadline = t.Deadline,
                creatorId = t.CreatorId,
                assigneUserId = t.AssigneUserId
            })
            .FirstOrDefaultAsync();

        if (task == null)
        {
            return NotFound(new { message = "Задача не найдена." });
        }

        return Ok(task);
    }

    [HttpPut("{taskId}")]
    public async Task<IActionResult> UpdateTask(
        int taskId,
        UpdateTaskDto dto)
    {
        var task = await _context.Tasks
            .FirstOrDefaultAsync(t => t.IdTask == taskId);

        if (task == null)
        {
            return NotFound(new { message = "Задача не найдена." });
        }

        if (dto.Status != "Новая" &&
            dto.Status != "В работе" &&
            dto.Status != "Выполнена")
        {
            return BadRequest(new
            {
                message = "Статус должен быть: Новая, В работе или Выполнена."
            });
        }

        if (dto.Priority != "Низкий" &&
            dto.Priority != "Средний" &&
            dto.Priority != "Высокий")
        {
            return BadRequest(new
            {
                message = "Приоритет должен быть: Низкий, Средний или Высокий."
            });
        }

        task.Name = dto.Name;
        task.Describe = dto.Describe;
        task.Status = dto.Status;
        task.Priority = dto.Priority;

        if (dto.Deadline.HasValue)
        {
            task.Deadline = dto.Deadline.Value;
        }

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Задача обновлена.",
            taskId = task.IdTask,
            name = task.Name,
            describe = task.Describe,
            status = task.Status,
            priority = task.Priority,
            deadline = task.Deadline
        });
    }

    [HttpDelete("{taskId}")]
    public async Task<IActionResult> DeleteTask(int taskId)
    {
        var task = await _context.Tasks
    .Include(t => t.Users)
    .FirstOrDefaultAsync(t => t.IdTask == taskId);

        if (task == null)
        {
            return NotFound(new { message = "Задача не найдена." });
        }

        task.Users.Clear();

        _context.Tasks.Remove(task);

        await _context.SaveChangesAsync();

        return Ok(new
        {
            message = "Задача удалена.",
            taskId
        });
    }
}

using ASKART.API.Data;
using Microsoft.AspNetCore.SignalR;
using Microsoft.EntityFrameworkCore;

namespace ASKART.API.Hubs;

public class ChatHub : Hub
{
    private readonly AskartContext _context;

    public ChatHub(AskartContext context)
    {
        _context = context;
    }

    public async Task JoinChat(int chatId, int userId)
    {
        var isMember = await _context.Chatmembers
            .AnyAsync(cm =>
                cm.ChatId == chatId &&
                cm.UserId == userId);

        if (!isMember)
        {
            throw new HubException(
                "Пользователь не состоит в этом чате.");
        }

        await Groups.AddToGroupAsync(
            Context.ConnectionId,
            $"chat_{chatId}");
    }

    public async Task LeaveChat(int chatId)
    {
        await Groups.RemoveFromGroupAsync(
            Context.ConnectionId,
            $"chat_{chatId}");
    }
}
