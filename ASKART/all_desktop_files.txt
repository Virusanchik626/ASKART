import os
from pathlib import Path

# Директория, где запускается скрипт
BASE_DIR = Path.cwd()

# Папки, из которых нужно собирать .cs и .xaml.cs файлы
TARGET_DIRS = ["MockData", "Models", "Resources", "Services", "Windows"]

# Файл для результата
OUTPUT_FILE = BASE_DIR / "all_api_files.txt"

# Расширения, которые ищем в подпапках
TARGET_EXTENSIONS = (".cs", ".xaml.cs")


def is_target_file(filename: str) -> bool:
    """Проверяет, что файл имеет расширение .cs или .xaml.cs"""
    return filename.endswith(TARGET_EXTENSIONS)


def collect_files():
    """Собирает список файлов для обработки."""
    files_to_process = []

    # 1. Все файлы в основной директории (без подпапок)
    for entry in sorted(BASE_DIR.iterdir()):
        if entry.is_file() and entry.name != OUTPUT_FILE.name:
            files_to_process.append(entry)

    # 2. .cs и .xaml.cs файлы из указанных папок (рекурсивно)
    for dir_name in TARGET_DIRS:
        dir_path = BASE_DIR / dir_name
        if not dir_path.is_dir():
            print(f"[!] Папка не найдена: {dir_name}")
            continue

        for root, dirs, files in os.walk(dir_path):
            # Сортируем для детерминированного порядка
            dirs.sort()
            for fname in sorted(files):
                if is_target_file(fname):
                    files_to_process.append(Path(root) / fname)

    # Убираем дубликаты с сохранением порядка
    seen = set()
    unique_files = []
    for f in files_to_process:
        if f not in seen:
            seen.add(f)
            unique_files.append(f)

    return unique_files


def write_files_to_output(files):
    """Записывает содержимое файлов в итоговый текстовый файл."""
    total = len(files)

    with OUTPUT_FILE.open("w", encoding="utf-8") as out:
        out.write(f"# Собрано файлов: {total}\n")
        out.write(f"# Базовая директория: {BASE_DIR}\n")
        out.write("=" * 80 + "\n\n")

        for idx, file_path in enumerate(files, 1):
            try:
                rel_path = file_path.relative_to(BASE_DIR)
            except ValueError:
                rel_path = file_path

            print(f"[{idx}/{total}] Обработка: {rel_path}")

            out.write(f"=== ФАЙЛ: {rel_path} ===\n")

            try:
                with file_path.open("r", encoding="utf-8") as f:
                    content = f.read()
            except UnicodeDecodeError:
                # Пробуем другую кодировку (например, Windows-1251)
                try:
                    with file_path.open("r", encoding="cp1251") as f:
                        content = f.read()
                    out.write("(кодировка: cp1251)\n")
                except Exception as e:
                    content = f"<<Ошибка чтения файла: {e}>>\n"
            except Exception as e:
                content = f"<<Ошибка чтения файла: {e}>>\n"

            out.write(content)
            if not content.endswith("\n"):
                out.write("\n")
            out.write("\n" + "-" * 80 + "\n\n")

    print(f"\n[✓] Готово! Результат сохранён в: {OUTPUT_FILE}")


def main():
    files = collect_files()
    if not files:
        print("Не найдено файлов для обработки.")
        return
    write_files_to_output(files)


if __name__ == "__main__":
    main()

using System.Windows;
using System.Windows.Media;
using Wpf.Ui.Appearance;

namespace Askart
{
    public static class ThemeHelper
    {
        private static bool started = false;

        private static string[] colorNames =
        {
            "Background", "SecondaryBackground", "PrimaryText", "SecondaryText",
            "InactiveText", "Active", "Success", "Error", "CardBackground",
            "InputBackground", "Line", "PinnedBarBackground", "PinnedBarText"
        };

        public static void Start()
        {
            if (started)
            {
                return;
            }
            started = true;

            ApplicationThemeManager.ApplySystemTheme();

            ApplyOurBrushes();

            ApplicationThemeManager.Changed += OnThemeChanged;
        }

        public static void Watch(Window window)
        {
            SystemThemeWatcher.Watch(window, Wpf.Ui.Controls.WindowBackdropType.None, false);
        }

        private static void OnThemeChanged(ApplicationTheme currentTheme, Color systemAccent)
        {
            ApplyOurBrushes();
        }

        private static void ApplyOurBrushes()
        {
            string prefix = "Light";
            if (ApplicationThemeManager.GetAppTheme() == ApplicationTheme.Dark)
            {
                prefix = "Dark";
            }

            foreach (string name in colorNames)
            {
                
                Color color = (Color)Application.Current.FindResource(prefix + name + "Color");

                
                
                Application.Current.Resources[name + "Brush"] = new SolidColorBrush(color);
            }
        }
    }
}

        public string GetUserRole(int userId)
        {
            foreach (ChatMember member in Members)
            {
                if (member.UserId == userId)
                {
                    return member.Role;
                }
            }
            return "Участник";
        }

        public bool IsAdmin(int userId)
        {
            string role = GetUserRole(userId);
            if (role == "Владелец" || role == "Администратор")
            {
                return true;
            }
            return false;
        }

        {
            if (TypeChat == "Личный" && Members.Count != 2)
            {
                return "Личный чат должен содержать ровно 2 участников.";

            if (TypeChat == "Группа" && Members.Count < 3)
            {
                return "Групповой чат должен содержать не менее 3 участников.";
            }
            return null;
        }

using System.Text.RegularExpressions;

namespace Askart.Models
{
    public class User
    {
        public int Id { get; set; }
        public string Surname { get; set; }
        public string Name { get; set; }
        public string Lastname { get; set; }
        public string Login { get; set; }
        public string Password { get; set; }
        public string Number { get; set; }
        public DateTime RegistrationDate { get; set; }
        public string Email { get; set; }
        public byte[] Avatar { get; set; }

        
        public string FullName
        {
            get { return (Surname + " " + Name + " " + Lastname).Trim(); }
        }

        public static bool IsValidEmail(string email)
        {
            if (string.IsNullOrWhiteSpace(email))
            {
                return false;
            }
            return Regex.IsMatch(email, @"^[^@\s]+@[^@\s]+\.[^@\s]+$");
        }

        public static bool IsValidPhone(string phone)
        {
            if (string.IsNullOrWhiteSpace(phone))
            {
                return false;
            }
            string clean = phone.Replace(" ", "").Replace("-", "").Replace("(", "").Replace(")", "");
            return Regex.IsMatch(clean, @"^\+\d{10,15}$");
        }

        public static bool IsValidPassword(string password)
        {
            if (string.IsNullOrWhiteSpace(password))
            {
                return false;
            }
            return password.Length >= 8;
        }
    }
}

        {
            
            foreach (User user in DataInitializer.Users)
            {
                if (user.Login == email && user.Password == password)
                {
                    return user;
                }
            }

        public static string Register(string name, string surname, string email, string password, string phone)
        {
            
            foreach (User user in DataInitializer.Users)
            {
                if (user.Login == email || user.Email == email)
                {
                    return "Пользователь с такой почтой уже существует";
                }
            }

            int newId = 1;
            foreach (User user in DataInitializer.Users)
            {
                if (user.Id >= newId)
                {
                    newId = user.Id + 1;
                }
            }

            User newUser = new User();
            newUser.Id = newId;
            newUser.Name = name;
            newUser.Surname = surname;
            newUser.Lastname = "";
            newUser.Login = email;
            newUser.Password = password;
            newUser.Email = email;
            newUser.Number = phone;
            newUser.RegistrationDate = DateTime.Now;

            DataInitializer.Users.Add(newUser);
            return null;
        }

                {
                    continue;
                }

                if (chat.IsArchived && showArchived == false)
                {
                    continue;
                }

                if (search != "" && chat.Name.ToLower().Contains(search) == false)
                {
                    continue;
                }

                result.Add(chat);
            }
            return result;
        }

        public static Chat CreateChat(string name, bool isGroup, bool isWork, User owner)
        {
            
            int newId = 1;
            foreach (Chat c in DataInitializer.Chats)
            {
                if (c.Id >= newId)

                    newId = c.Id + 1;
                }
            }

            Chat newChat = new Chat();
            newChat.Id = newId;
            newChat.Name = name;
            newChat.CreatedAt = DateTime.Now;
            newChat.OwnerId = owner.Id;

            if (isGroup)
            {
                newChat.TypeChat = "Группа";
                if (isWork)
                {
                    newChat.TypeGroup = "Рабочая";
                }
                else
                {
                    newChat.TypeGroup = "Личная";
                }
            }
            else

                newChat.TypeGroup = null;
            }

            ChatMember me = new ChatMember();
            me.ChatId = newId;
            me.UserId = owner.Id;
            me.Role = "Владелец";
            newChat.Members.Add(me);

        {

            {
                if (column.Name == name)
                {
                    return "Колонка с таким названием уже существует";
                }
            }

            KanbanColumn newColumn = new KanbanColumn();
            newColumn.Name = name;
            newColumn.Color = "#9E9E9E";
            chat.KanbanColumns.Add(newColumn);
            return null;
        }

        
        public static void DeleteColumn(Chat chat, KanbanColumn column)
        {
            
            List<TaskItem> tasksToRemove = new List<TaskItem>();
            foreach (TaskItem task in chat.Tasks)
            {
                if (task.Status == column.Name)
                {
                    tasksToRemove.Add(task);
                }
            }
            foreach (TaskItem task in tasksToRemove)
            {
                chat.Tasks.Remove(task);
            }

            chat.KanbanColumns.Remove(column);
        }

        
        public static void MoveColumn(Chat chat, KanbanColumn column, int direction)
        {
            
            int oldIndex = chat.KanbanColumns.IndexOf(column);
            int newIndex = oldIndex + direction;

            if (oldIndex < 0 || newIndex < 0 || newIndex >= chat.KanbanColumns.Count)
            {
                return;
            }

            chat.KanbanColumns.RemoveAt(oldIndex);
            chat.KanbanColumns.Insert(newIndex, column);
        }

        
        public static string UpdateColumn(Chat chat, KanbanColumn column, string newName, string newColor)
        {
            
            if (newName != column.Name)
            {
                foreach (KanbanColumn other in chat.KanbanColumns)
                {
                    if (other != column && other.Name == newName)
                    {
                        return "Колонка с таким названием уже существует";
                    }
                }

                
                foreach (TaskItem task in chat.Tasks)
                {
                    if (task.Status == column.Name)
                    {
                        task.Status = newName;
                    }
                }
            }

            column.Name = newName;
            column.Color = newColor;
            return null;
        }

        private void ShowTopSenders(List<Message> messages)
        {
            List<string> names = new List<string>();
            List<int> counts = new List<int>();

            foreach (Message message in messages)
            {
                int index = names.IndexOf(message.SenderName);
                if (index == -1)
                {
                    names.Add(message.SenderName);
                    counts.Add(1);
                }
                else
                {
                    counts[index] = counts[index] + 1;
                }
            }

            
            for (int i = 0; i < names.Count - 1; i++)
            {
                for (int j = 0; j < names.Count - 1 - i; j++)
                {
                    if (counts[j] < counts[j + 1])
                    {
                        int tempCount = counts[j];
                        counts[j] = counts[j + 1];
                        counts[j + 1] = tempCount;

                        string tempName = names[j];
                        names[j] = names[j + 1];
                        names[j + 1] = tempName;
                    }
                }
            }

            for (int i = 0; i < names.Count && i < 5; i++)
            {
                int rank = i + 1;
                StatsPanel.Children.Add(CreateInfoRow("#" + rank + " " + names[i], counts[i] + " сообщ."));
            }
        }

            {

        private Border CreateMessageBlock(Message message)
        {
            bool isOwn = (message.SenderId == App.CurrentUser.Id);
            bool isAdmin = currentChat.IsAdmin(App.CurrentUser.Id);

            Border border = new Border();
            border.Margin = new Thickness(0, 5, 0, 5);
            border.Padding = new Thickness(15, 10, 15, 10);
            border.CornerRadius = new CornerRadius(12);
            border.MaxWidth = 500;

            if (isOwn)
            {
                border.HorizontalAlignment = HorizontalAlignment.Right;
                border.SetResourceReference(Border.BackgroundProperty, "ActiveBrush");
            }
            else
            {
                border.HorizontalAlignment = HorizontalAlignment.Left;
                border.SetResourceReference(Border.BackgroundProperty, "SecondaryBackgroundBrush");
            }

            if (message.IsPinned)
            {
                border.BorderBrush = new SolidColorBrush(Color.FromRgb(255, 193, 7));
                border.BorderThickness = new Thickness(2);
            }

            StackPanel stackPanel = new StackPanel();

            
            Grid headerGrid = new Grid();
            headerGrid.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(1, GridUnitType.Star) });
            headerGrid.ColumnDefinitions.Add(new ColumnDefinition { Width = GridLength.Auto });

            string senderText = message.SenderName;
            if (message.IsPinned)
            {
                senderText = senderText + " 📌";
            }

            nameText.FontWeight = FontWeights.SemiBold;
            nameText.Margin = new Thickness(0, 0, 0, 5);
            nameText.VerticalAlignment = VerticalAlignment.Center;
            if (isOwn)
            {
                nameText.Foreground = Brushes.White;
            }
            Grid.SetColumn(nameText, 0);
            headerGrid.Children.Add(nameText);

            if (isOwn || isAdmin)
            {
                StackPanel actionsPanel = new StackPanel();
                actionsPanel.Orientation = Orientation.Horizontal;
                actionsPanel.VerticalAlignment = VerticalAlignment.Center;

                if (isOwn)
                {
                    Button editButton = CreateActionButton("✏️", "Редактировать", message, isOwn, "SecondaryTextBrush");
                    editButton.Click += EditMessage_Click;
                    actionsPanel.Children.Add(editButton);
                }

                string deleteTip = "Удалить";
                if (isAdmin && isOwn == false)
                {
                    deleteTip = "Удалить (админ)";
                }
                Button deleteButton = CreateActionButton("✕", deleteTip, message, isOwn, "ErrorBrush");
                deleteButton.Click += DeleteMessage_Click;
                actionsPanel.Children.Add(deleteButton);

                if (isAdmin)
                {
                    string pinIcon = "📌";
                    string pinTip = "Закрепить";
                    if (message.IsPinned)
                    {
                        pinIcon = "📍";
                        pinTip = "Открепить";
                    }
                    Button pinButton = CreateActionButton(pinIcon, pinTip, message, isOwn, "SecondaryTextBrush");
                    pinButton.Click += TogglePin_Click;
                    actionsPanel.Children.Add(pinButton);
                }

                Button replyButton = CreateActionButton("↩", "Ответить", message, isOwn, "SecondaryTextBrush");
                replyButton.Click += Reply_Click;
                actionsPanel.Children.Add(replyButton);

                Grid.SetColumn(actionsPanel, 1);
                headerGrid.Children.Add(actionsPanel);
            }

            stackPanel.Children.Add(headerGrid);

            
            if (message.ReplyId.HasValue)
            {
                Message replied = DataService.GetMessageById(currentChat, message.ReplyId.Value);
                if (replied != null)
                {
                    Border replyBorder = new Border();
                    replyBorder.Background = new SolidColorBrush(Color.FromArgb(40, 255, 255, 255));
                    replyBorder.CornerRadius = new CornerRadius(6);
                    replyBorder.Padding = new Thickness(8, 6, 8, 6);
                    replyBorder.Margin = new Thickness(0, 5, 0, 5);
                    replyBorder.BorderBrush = new SolidColorBrush(Color.FromRgb(100, 200, 255));
                    replyBorder.BorderThickness = new Thickness(2, 0, 0, 0);

                    TextBlock replyText = UiHelper.MakeText("↩ " + replied.SenderName + ": " + replied.Text, 11, "SecondaryTextBrush");
                    replyText.FontStyle = FontStyles.Italic;
                    replyText.TextWrapping = TextWrapping.Wrap;
                    replyText.MaxHeight = 40;
                    if (isOwn)
                    {
                        replyText.Foreground = new SolidColorBrush(Color.FromRgb(200, 240, 255));
                    }

                    replyBorder.Child = replyText;
                    stackPanel.Children.Add(replyBorder);
                }
            }

            
            TextBlock messageText = UiHelper.MakeText(message.Text, 14, "PrimaryTextBrush");
            messageText.TextWrapping = TextWrapping.Wrap;
            if (isOwn)
            {
                messageText.Foreground = Brushes.White;
            }
            stackPanel.Children.Add(messageText);

            
            string timeString = message.SendTime.ToString("HH:mm");
            if (message.EditTime.HasValue)
            {
                timeString = timeString + " (ред.)";
            }
            TextBlock timeText = UiHelper.MakeText(timeString, 11, "InactiveTextBrush");
            timeText.HorizontalAlignment = HorizontalAlignment.Right;
            timeText.Margin = new Thickness(0, 5, 0, 0);
            if (isOwn)
            {
                timeText.Foreground = new SolidColorBrush(Color.FromRgb(200, 240, 255));
            }
            stackPanel.Children.Add(timeText);

            border.Child = stackPanel;

            
            ContextMenu contextMenu = new ContextMenu();

            MenuItem copyItem = CreateMenuItem("Копировать", message);
            copyItem.Click += CopyMessage_Click;
            contextMenu.Items.Add(copyItem);

            MenuItem replyItem = CreateMenuItem("↩ Ответить", message);
            replyItem.Click += Reply_Click;
            contextMenu.Items.Add(replyItem);

            if (isOwn)
            {
                MenuItem editItem = CreateMenuItem("✏️ Редактировать", message);
                editItem.Click += EditMessage_Click;
                contextMenu.Items.Add(editItem);
            }

            if (isOwn || isAdmin)
            {
                MenuItem deleteItem = CreateMenuItem("✕ Удалить", message);
                deleteItem.Click += DeleteMessage_Click;
                contextMenu.Items.Add(deleteItem);
            }

            if (isAdmin)
            {
                string pinHeader = "📌 Закрепить";
                if (message.IsPinned)
                {
                    pinHeader = "📍 Открепить";
                }
                MenuItem pinItem = CreateMenuItem(pinHeader, message);
                pinItem.Click += TogglePin_Click;
                contextMenu.Items.Add(pinItem);
            }

            border.ContextMenu = contextMenu;
            return border;
        }

        private void DeleteMessage(Message message)
        {
            bool isOwn = (message.SenderId == App.CurrentUser.Id);
            bool isAdmin = currentChat.IsAdmin(App.CurrentUser.Id);

            if (isOwn == false && isAdmin == false)
            {
                return;
            }

            string confirmText = "Удалить ваше сообщение?";
            if (isOwn == false)
            {
                confirmText = "Удалить сообщение пользователя " + message.SenderName + "? (права администратора)";
            }

            MessageBoxResult result = MessageBox.Show(confirmText, "Удаление сообщения",
                MessageBoxButton.YesNo, MessageBoxImage.Question);

            if (result == MessageBoxResult.Yes)
            {
                DataService.DeleteMessage(currentChat, message);
                LoadMessages();
                UpdatePinnedBar();
            }
        }

        private void TogglePinMessage(Message message)
        {
            if (currentChat.IsAdmin(App.CurrentUser.Id) == false)
            {
                return;
            }

            DataService.SetPinned(message, !message.IsPinned);
            LoadMessages();
            UpdatePinnedBar();

            if (message.IsPinned)
            {
                MessageBox.Show("Сообщение закреплено 📌", "Закрепление",
                    MessageBoxButton.OK, MessageBoxImage.Information);
            }
            else
            {
                MessageBox.Show("Сообщение откреплено", "Закрепление",
                    MessageBoxButton.OK, MessageBoxImage.Information);
            }
        }

        private void ReplyToMessage(Message message)

            ReplyPreviewText.Text = preview;
            ReplyPanel.Visibility = Visibility.Visible;

            MessageTextBox.Focus();
        }

        {

            LoadMessages();
        }

using Askart.Models;

            InitializeComponent();
            this.chat = chat;
            ChatNameText.Text = chat.Name;
            LoadPinnedMessages();
        }

        private void LoadPinnedMessages()
        {
            PinnedPanel.Children.Clear();

            List<Message> pinnedMessages = DataService.GetPinnedMessages(chat);

            if (pinnedMessages.Count == 0)
            {
                TextBlock emptyText = UiHelper.MakeText("В этом чате пока нет закреплённых сообщений", 14, "SecondaryTextBrush");
                emptyText.HorizontalAlignment = HorizontalAlignment.Center;
                emptyText.Margin = new Thickness(0, 30, 0, 0);
                PinnedPanel.Children.Add(emptyText);
                return;
            }

            foreach (Message message in pinnedMessages)
            {
                PinnedPanel.Children.Add(CreatePinnedCard(message));
            }
        }

        private Border CreatePinnedCard(Message message)
        {
            Border border = new Border();
            border.SetResourceReference(Border.BackgroundProperty, "CardBackgroundBrush");
            border.CornerRadius = new CornerRadius(12);
            border.Padding = new Thickness(15);

            border.BorderThickness = new Thickness(3);

            DropShadowEffect shadow = new DropShadowEffect();
            shadow.Color = Colors.Black;
            shadow.BlurRadius = 10;
            shadow.ShadowDepth = 2;
            shadow.Opacity = 0.15;
            border.Effect = shadow;

            StackPanel stackPanel = new StackPanel();

            Grid headerGrid = new Grid();
            headerGrid.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(1, GridUnitType.Star) });
            headerGrid.ColumnDefinitions.Add(new ColumnDefinition { Width = GridLength.Auto });

using Askart.Models;
using Askart.Services;
using System.Windows;
using System.Windows.Input;

namespace Askart.Windows
{
    public partial class RegisterWindow : Window
    {
        public RegisterWindow()
        {
            ThemeHelper.Watch(this);
            InitializeComponent();
        }

        private void Register_Click(object sender, RoutedEventArgs e)
        {
            string firstName = FirstNameTextBox.Text.Trim();
            string lastName = LastNameTextBox.Text.Trim();
            string email = EmailTextBox.Text.Trim();
            string password = PasswordBox.Password;
            string phone = PhoneTextBox.Text.Trim();

            
            if (string.IsNullOrWhiteSpace(firstName) ||
                string.IsNullOrWhiteSpace(lastName) ||
                string.IsNullOrWhiteSpace(email) ||
                string.IsNullOrWhiteSpace(password))
            {
                MessageBox.Show("Пожалуйста, заполните все обязательные поля", "Ошибка",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            
            if (User.IsValidEmail(email) == false)
            {
                MessageBox.Show("Неверный формат почты. Пример: user@example.com", "Ошибка",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            
            if (User.IsValidPassword(password) == false)
            {
                MessageBox.Show("Пароль должен содержать минимум 8 символов", "Ошибка",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            
            if (phone != "" && User.IsValidPhone(phone) == false)
            {
                MessageBox.Show("Неверный формат телефона. Пример: +79991234567", "Ошибка",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            
            string error = DataService.Register(firstName, lastName, email, password, phone);

            if (error == null)
            {
                MessageBox.Show("Регистрация успешна! Теперь вы можете войти.", "Успех",
                    MessageBoxButton.OK, MessageBoxImage.Information);

                LoginWindow loginWindow = new LoginWindow();
                loginWindow.Show();
                this.Close();
            }
            else
            {
                MessageBox.Show(error, "Ошибка", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void Login_Click(object sender, MouseButtonEventArgs e)
        {
            LoginWindow loginWindow = new LoginWindow();
            loginWindow.Show();
            this.Close();
        }
    }
}
